def Schubert.RS.Representation.HasOrderedPBWBasis : ℕ → Prop :=
fun n => ∀ (order : RootOrdering n), ∃ b, ∀ (powers : PositiveRoot n → ℕ), b powers = orderedRootMonomial order powers
def Schubert.RS.Representation.orderedRootMonomial : {n : ℕ} → RootOrdering n → (PositiveRoot n → ℕ) → Enveloping n :=
fun {n} order powers =>
  (List.map (fun r => (UniversalEnvelopingAlgebra.ι ℂ) ((rootBasis n) r) ^ powers r) order.roots).prod
@[reducible] def Schubert.RS.Representation.Enveloping : ℕ → Type :=
fun n => UniversalEnvelopingAlgebra ℂ ↥(upperNilpotent n)
def Schubert.RS.Representation.orderedRootBasis : {n : ℕ} →
  RootOrdering n → Module.Basis (PositiveRoot n → ℕ) ℂ (Enveloping n) :=
fun {n} order => Module.Basis.mk ⋯ ⋯
theorem Schubert.RS.Representation.orderedPBWBasis_exists : ∀ (n : ℕ), HasOrderedPBWBasis n :=
fun n order => Exists.intro (orderedRootBasis order) (orderedRootBasis_apply order)
